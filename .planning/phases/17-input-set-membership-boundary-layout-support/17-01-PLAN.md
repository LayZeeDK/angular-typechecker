```typescript
// filter-diagnostics.ts

export interface FilterOptions {
  basePath: string;               // solution/host tsconfig dir (narrowed base clause, D-04a)
  includeDeps: boolean;
  useCaseSensitiveFileNames: boolean;
  realpath: (filePath: string) => string;
  // NEW (D-02): the DECLARED rootName .ts paths (raw, pre-canonical) whose union
  // is the input set. Empty is legal (guard paths omit the filter entirely).
  inputTs: readonly string[];
}

export interface FilterResult {
  kept: ts.Diagnostic[];
  // D-05/D-07: REPLACES the old scalar `suppressedCount` (do NOT keep it).
  suppressedThirdParty: number;        // node_modules segment suppressions (quiet)
  suppressedInGraphErrorCount: number; // suppressed first-party Error diagnostics
  suppressedInGraphWarningCount: number; // suppressed first-party Warning diagnostics
  suppressedInGraphFiles: readonly string[]; // distinct canonical files of the above (advisory)
}

// Pure boolean; reads ONLY public ts.Diagnostic fields. Zero ngtsc internals.
export function keep(
  diagnostic: ts.Diagnostic,
  inputSet: ReadonlySet<string>,   // dual-identity: contains BOTH raw and full canonical rootNames
  options: {
    canonicalizeRaw: (p: string) => string;            // slash + case-fold, NO realpath, never undefined
    canonicalizeFull: (p: string) => string | undefined; // realpath + slash + case-fold (existing createCanonicalizer)
    canonicalBase: string | undefined;                 // full-canonical solution/host base
    includeDeps: boolean;
  },
): boolean;

// NEW raw canonicalizer (no realpath). Same case-fold policy as createCanonicalizer.
export function createRawCanonicalizer(
  options: Pick<FilterOptions, 'useCaseSensitiveFileNames'>,
): (filePath: string) => string;
```

It lives under `src/core/**`, so the existing D-11 eslint block ALREADY bans `@nx/devkit`, `@nx/*`,

```typescript
// MOVED from init/generator.ts (unchanged string).

  'angular-typechecker: Angular CLI has no build/target-result cache to seed, so ' +
  'the typecheck target(s) were wired without caching. On an Nx workspace, target ' +
  'caching is configured automatically.';

// MOVED from generators/ng-add/generator.ts (unchanged string).
export const NO_ANGULAR_JSON_NOTICE =
  'angular-typechecker: no angular.json found, so no typecheck targets were ' +
  'wired. angular-typechecker was ensured as a devDependency. Run `ng add ' +
  'angular-typechecker` from an Angular CLI workspace to auto-wire targets, or ' +
  'wire a single project with `nx g angular-typechecker:configuration <project>`.';

// MOVED from configuration/generator.ts (the raw on-disk angular.json shapes; architect map).
export interface AngularJsonTarget { builder?: string; options?: Record<string, unknown>; [key: string]: unknown; }
export interface AngularJsonProject { projectType?: 'application' | 'library'; root?: string; architect?: Record<string, AngularJsonTarget>; [key: string]: unknown; }
export interface AngularJsonWorkspace { projects: Record<string, AngularJsonProject>; [key: string]: unknown; }

// Pure: default + empty/whitespace guard. Byte-preserve the message.
// Error: `--targetName for project "${project}" must be a non-empty target name. Omit it to use the default "typecheck".`
export function resolveTargetName(targetName: string | undefined, project: string): string;

// Pure: the OQ-1 override rung (absolute passes through; relative is project-root-joined + exists-probed).
// Error: `--tsConfig "${tsConfig}" for project "${project}" resolves to "${overridePath}", which does not exist. Pass a path relative to the project root (or an absolute path).`
export function resolveTsConfigOverride(root: string, tsConfig: string, project: string, exists: (path: string) => boolean): string;

// Pure: the RF-01/Approach-A [build, spec] leaf ARRAY. override -> single-element [resolved];
// else projectType build leaf (application -> tsconfig.app.json, else tsconfig.lib.json) + tsconfig.spec.json,
// each posix.join(root, leaf) then exists-probed; missing dropped; empty -> throw.
// Error: `Could not resolve a tsconfig for project "${project}": no "${buildLeaf}" and no "${specLeaf}". Pass --tsConfig explicitly.`
export function resolveTsConfigLeaves(root: string, projectType: 'application' | 'library' | undefined, tsConfigOverride: string | undefined, project: string, exists: (path: string) => boolean): string[];

// Pure: MUTATES workspace.projects[project].architect[targetName]. Collision by BUILDER id, idempotent merge.
// Error: `Project "${project}" already has a "${targetName}" target using builder "${existing.builder}". Choose a different --targetName or remove the existing target.`
export function wireTypecheckTarget(workspace: AngularJsonWorkspace, project: string, targetName: string, tsConfig: string[]): void;
```
